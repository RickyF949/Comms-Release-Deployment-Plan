# syntax=docker/dockerfile:1
FROM python:3.13-slim-bookworm

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PIP_NO_CACHE_DIR=1 \
    HOME=/home/comms

RUN apt-get update \
    && apt-get install -y --no-install-recommends ca-certificates curl unzip git build-essential \
    && rm -rf /var/lib/apt/lists/* \
    && groupadd --gid 10001 comms \
    && useradd --uid 10001 --gid comms --create-home comms

WORKDIR /srv/comms
COPY requirements.txt ./requirements.txt
RUN pip install --no-cache-dir -r requirements.txt

# Explicit allowlist: never COPY the repository's .env or local credentials.
COPY --chown=comms:comms rxconfig.py ./rxconfig.py
COPY --chown=comms:comms app/ ./app/
COPY --chown=comms:comms assets/ ./assets/
COPY --chown=comms:comms reflex.lock/ ./reflex.lock/
COPY --chown=comms:comms db_migrations/ ./db_migrations/
RUN chown comms:comms /srv/comms

USER comms
EXPOSE 3000 8000
STOPSIGNAL SIGTERM
CMD ["reflex", "run", "--env", "prod", "--frontend-port", "3000", "--backend-port", "8000", "--backend-host", "0.0.0.0"]
